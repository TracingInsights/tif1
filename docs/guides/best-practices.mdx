### Use selective data loading

Control what data gets fetched to optimize performance. Load only the necessary data.

<CodeGroup>
```python Good - Selective Loading
import tif1

# Load only laps data, skip telemetry/weather/messages
session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
# session.load() is optional: session.laps fetches only the laps on first access
laps = session.laps  # Fast - only laps fetched
```

```python Avoid - Loading Everything
session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
session.load()  # Fetches laps, weather, messages, and the telemetry of every lap
laps = session.laps  # Slower - unnecessary data fetched
```
</CodeGroup>

### Batch telemetry fetching

Never fetch telemetry in a loop. Use batch methods instead.

<CodeGroup>
```python Good - Batch Fetching
# One parallel fetch for every driver's fastest lap
session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
tels = session.get_fastest_laps_tels(by_driver=True)

# tels is one DataFrame with all drivers' fastest lap telemetry (not a dict)
for driver, driver_tel in tels.groupby("Driver"):
    print(f"{driver}: {driver_tel['Speed'].max()} km/h")
```

```python Avoid - Sequential Loop
# Very slow - fetches one at a time
for driver in session.drivers_df["Driver"]:
    driver_obj = session.get_driver(driver)
    fastest = driver_obj.get_fastest_lap()
    tel = driver_obj.get_lap(int(fastest["LapNumber"].iloc[0])).telemetry
    print(f"{driver}: {tel['Speed'].max()} km/h")
```
</CodeGroup>

### Choose the right lib

Use polars for large datasets and performance-critical code. `lib="polars"` returns polars DataFrames from `session.laps`, `session.weather`, and the telemetry accessors. `session.drivers_df` stays a pandas DataFrame. Without polars, the session falls back to pandas.

```python
import tif1

# For large datasets or batch processing
session = tif1.get_session(
    2024,
    "Belgian Grand Prix",
    "Race",
    lib="polars"  # polars DataFrames instead of pandas
)
```

### Keep caching enabled

Never disable caching unless absolutely necessary.

<CodeGroup>
```python Good - Cache Enabled (Default)
session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
# Subsequent loads come from the local SQLite cache instead of the CDN
```

```python Avoid - Cache Disabled
session = tif1.get_session(2024, "Belgian Grand Prix", "Race", enable_cache=False)
# Every load hits the CDN
```
</CodeGroup>

## Code quality best practices

### Use Type Hints

tif1 ships type hints for the public API. Use them for better IDE support.

```python
from tif1 import Session
import tif1

def analyze_session(session: Session) -> dict[str, float]:
    """Analyze session and return metrics."""
    laps = session.laps  # Annotated as a pandas or polars DataFrame

    # LapTime is a Timedelta; LapTimeSeconds is the float the annotation promises
    return {
        "total_laps": len(laps),
        "avg_lap_time": laps["LapTimeSeconds"].mean(),
    }

session: Session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
metrics = analyze_session(session)
```
### Handle errors gracefully

Always handle tif1 exceptions appropriately.

```python
import tif1
from tif1 import DataNotFoundError, NetworkError, InvalidDataError

def safe_load_session(year: int, gp: str, session: str):
    """Load session with proper error handling."""
    try:
        loaded = tif1.get_session(year, gp, session)
        loaded.laps  # get_session is lazy: the fetch (and its errors) happen here
        return loaded
    except DataNotFoundError:
        print(f"Data not found for {year} {gp} {session}")
        return None
    except NetworkError as e:
        print(f"Network error: {e}")
        # Maybe retry or use cached data
        return None
    except InvalidDataError as e:
        print(f"Data corruption: {e}")
        # Drop cached payloads before a retry
        tif1.get_cache().clear()
        return None
```

### Use context managers for resources

When working with exports or databases, use context managers.

```python
import sqlite3

session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
laps = session.laps

# Good - the with block commits the transaction, or rolls it back on error
with sqlite3.connect("f1_data.db") as conn:
    # SQLite column names are case-insensitive: to_sql fails with "duplicate
    # column name: team" unless the lowercase "team" duplicate of "Team" is dropped
    laps_sql = laps.loc[:, ~laps.columns.str.lower().duplicated()]
    laps_sql.to_sql("laps", conn, if_exists="replace", index=False)
```

## Data analysis best practices

### Filter early, aggregate late

Filter data as early as possible to reduce memory usage.

<CodeGroup>
```python Good - Filter First
laps = session.laps

# Filter to relevant drivers first
top_3 = laps[laps["Driver"].isin(["VER", "HAM", "LEC"])]

# Then aggregate
avg_times = top_3.groupby("Driver")["LapTime"].mean()
```

```python Avoid - Load Everything
laps = session.laps

# Aggregate everything
all_avg = laps.groupby("Driver")["LapTime"].mean()

# Then filter (wasted computation)
top_3_avg = all_avg[all_avg.index.isin(["VER", "HAM", "LEC"])]
```
</CodeGroup>

### Clean data before analysis

Always filter out invalid laps before analysis.

```python
def clean_laps(laps):
    """Remove invalid laps for analysis."""
    # Remove deleted laps
    clean = laps[~laps["Deleted"]]

    # Remove pit laps
    clean = clean[clean["PitInTime"].isna()]

    # Remove outliers (> 7% slower than fastest)
    fastest = clean["LapTime"].min()
    clean = clean[clean["LapTime"] < fastest * 1.07]

    # Remove lap 1 (standing start)
    clean = clean[clean["LapNumber"] > 1]

    return clean

laps = session.laps
clean_laps = clean_laps(laps)
```

### Use Vectorized Operations

Avoid loops when working with DataFrames.

<CodeGroup>
```python Good - Vectorized
# Calculate lap time delta to fastest
fastest_time = laps["LapTime"].min()
laps["DeltaToFastest"] = laps["LapTime"] - fastest_time
```

```python Avoid - Loop
# Slow and inefficient
deltas = []
fastest_time = laps["LapTime"].min()
for _, lap in laps.iterrows():
    deltas.append(lap["LapTime"] - fastest_time)
laps["DeltaToFastest"] = deltas
```
</CodeGroup>

## Reliability best practices

### Check data availability first

Before processing multiple sessions, check which sessions each event has.

```python
def process_season(year: int):
    """Process all races in a season."""
    # get_event_schedule() returns a DataFrame; iterate the EventName column
    events = tif1.get_event_schedule(year, include_testing=False)["EventName"]

    for event in events:
        sessions = tif1.get_sessions(year, event)

        if "Race" not in sessions:
            print(f"No race data for {event}")
            continue

        try:
            session = tif1.get_session(year, event, "Race")
            print(f"{event}: {len(session.laps)} laps")
        except Exception as e:
            print(f"Failed to process {event}: {e}")
            continue
```
### Use circuit breaker awareness

Check circuit breaker status for long-running jobs.

```python
import tif1

def batch_process(year: int):
    """Process multiple sessions with circuit breaker awareness."""
    cb = tif1.get_circuit_breaker()

    if cb.state == "open":
        print("Circuit breaker is open, resetting...")
        tif1.reset_circuit_breaker()

    # Continue processing
    events = tif1.get_event_schedule(year, include_testing=False)["EventName"]
    for event in events:
        session = tif1.get_session(year, event, "Race")
        print(f"{event}: {len(session.laps)} laps")
```

### Enable debug logging for troubleshooting

When debugging, enable detailed logging.

```python
import tif1
import logging

# Enable debug logging
tif1.setup_logging(logging.DEBUG)

# Now all operations show detailed logs
session = tif1.get_session(2024, "Belgian Grand Prix", "Race")
```

## Memory management best practices

### Process data in chunks

For large-scale analysis, process drivers one at a time.

```python
def analyze_all_drivers(session):
    """Analyze one driver at a time."""
    results = []

    for driver_code in session.drivers_df["Driver"]:
        driver = session.get_driver(driver_code)
        laps = driver.laps

        # Analyze (LapTimeSeconds keeps the metrics as plain floats)
        result = {
            "driver": driver_code,
            "avg_time": laps["LapTimeSeconds"].mean(),
            "fastest": laps["LapTimeSeconds"].min(),
        }
        results.append(result)

        # Free memory
        del laps

    return results
```

### Use Polars for large datasets

Polars frames use less memory than the pandas frames of the same session. Pass `lib="polars"` per session.

```python
# For processing entire seasons
session = tif1.get_session(
    2024,
    "Belgian Grand Prix",
    "Race",
    lib="polars"
)
```

### Clear cache periodically

For long-running applications, free the in-memory cache tiers, or clear the cache to remove all stored data.

```python
import tif1

# At application startup or periodically
cache = tif1.get_cache()
cache.invalidate("memory")  # Frees the in-memory tiers; SQLite data stays
cache.clear()  # Deletes every SQLite and in-memory entry

# Or use the FastF1-compatible API
from tif1.fastf1_compat import Cache
Cache.clear_cache()  # Same as clear() on the active cache
```

## Testing best practices

### Mock network calls in tests

Do not hit the real CDN in unit tests.

```python
import pytest
from unittest.mock import patch
import tif1

def test_session_loading():
    """Test session loading with mocked network."""
    mock_data = {"drivers": [{"driver": "VER", "team": "Red Bull", "dn": 1}]}

    # Patch both fetch seams (retry and fast path) and skip the cache
    with (
        patch.object(tif1.Session, "_fetch_from_cdn", return_value=mock_data),
        patch.object(tif1.Session, "_fetch_from_cdn_fast", return_value=mock_data),
    ):
        session = tif1.get_session(2024, "Belgian Grand Prix", "Race", enable_cache=False)
        assert session.drivers_df["Driver"].tolist() == ["VER"]
```

### Use small datasets for tests

Test with minimal data to keep tests fast.

```python
def test_lap_filtering():
    """Test lap filtering logic."""
    # Use a small session (Practice 1) for faster tests
    session = tif1.get_session(2024, "Belgian Grand Prix", "Practice 1")
    laps = session.laps

    # Test filtering (LapTime is a Timedelta; compare LapTimeSeconds with a number)
    fast_laps = laps[laps["LapTimeSeconds"] < 110]
    assert len(fast_laps) >= 0
```

## Documentation best practices

### Document the analysis

Add docstrings and comments to the analysis code.

```python
def calculate_tire_degradation(laps, compound="SOFT"):
    """
    Calculate tire degradation rate for a specific compound.

    Args:
        laps: DataFrame with lap data
        compound: Tire compound to analyze (SOFT, MEDIUM, HARD)

    Returns:
        float: Degradation rate in seconds per lap

    Example:
        >>> laps = session.laps
        >>> deg = calculate_tire_degradation(laps, "SOFT")
        >>> print(f"Degradation: {deg:.3f}s/lap")
    """
    compound_laps = laps[laps["Compound"] == compound]

    # Linear regression on TyreLife vs LapTimeSeconds (scipy needs floats)
    from scipy import stats
    data = compound_laps[["TyreLife", "LapTimeSeconds"]].dropna()
    slope, _, _, _, _ = stats.linregress(
        data["TyreLife"],
        data["LapTimeSeconds"]
    )

    return slope
```
